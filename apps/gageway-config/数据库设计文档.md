# 网关配置管理系统 - 数据库设计文档

## 1. 系统概述

本系统设计用于管理多应用的路由配置，支持重定向和OSS映射两种配置类型，具备完整的权限控制和版本管理功能。

## 2. 核心模块

### 2.1 应用管理模块
- 应用基础信息管理
- 域名绑定管理
- 应用状态控制

### 2.2 路由配置模块
- 路由规则配置
- 重定向配置
- OSS映射配置
- 配置优先级管理

### 2.3 权限管理模块
- 用户权限控制
- 角色权限分配
- 应用级权限隔离

### 2.4 版本控制模块
- 配置变更记录
- 版本回滚支持
- 操作审计日志

## 3. 数据库表结构设计

### 3.1 应用管理表

```prisma
// 应用信息表
model Application {
  id          String            @id @default(cuid())
  name        String            // 应用名称
  code        String            @unique // 应用唯一标识码
  domain      String            @unique // 绑定域名
  description String?           // 应用描述
  status      ApplicationStatus @default(ACTIVE) // 应用状态
  
  // 关联关系
  routes      RouteConfig[]     // 路由配置
  permissions AppPermission[]   // 应用权限
  versions    ConfigVersion[]   // 配置版本
  
  createdAt   DateTime          @default(now())
  updatedAt   DateTime          @updatedAt
  createdBy   String?           // 创建者ID
  
  @@map("applications")
}

enum ApplicationStatus {
  ACTIVE      // 活跃
  INACTIVE    // 停用
  MAINTENANCE // 维护中
}
```

### 3.2 路由配置表

```prisma
// 路由配置主表
model RouteConfig {
  id            String          @id @default(cuid())
  applicationId String          // 应用ID
  name          String          // 配置名称
  path          String          // 路由路径
  method        HttpMethod      @default(GET) // HTTP方法
  configType    ConfigType      // 配置类型
  priority      Int             @default(0) // 优先级
  status        RouteStatus     @default(ACTIVE) // 状态
  
  // 配置详情（根据configType决定使用哪个）
  redirectConfig RedirectConfig? // 重定向配置
  ossConfig      OssConfig?      // OSS映射配置
  
  // 关联关系
  application   Application     @relation(fields: [applicationId], references: [id], onDelete: Cascade)
  versions      ConfigVersion[] // 版本记录
  
  createdAt     DateTime        @default(now())
  updatedAt     DateTime        @updatedAt
  createdBy     String?         // 创建者ID
  
  @@unique([applicationId, path, method]) // 同应用下路径+方法唯一
  @@map("route_configs")
}

// 重定向配置表
model RedirectConfig {
  id           String      @id @default(cuid())
  routeId      String      @unique // 路由配置ID
  targetUrl    String      // 目标URL
  statusCode   Int         @default(302) // HTTP状态码
  queryForward Boolean     @default(true) // 是否转发查询参数
  headers      Json?       // 自定义响应头
  
  route        RouteConfig @relation(fields: [routeId], references: [id], onDelete: Cascade)
  
  @@map("redirect_configs")
}

// OSS映射配置表
model OssConfig {
  id            String      @id @default(cuid())
  routeId       String      @unique // 路由配置ID
  bucket        String      // OSS存储桶
  objectKey     String      // 对象键（文件路径）
  region        String?     // 区域
  endpoint      String?     // 自定义端点
  cacheControl  String?     // 缓存控制
  contentType   String?     // 内容类型
  headers       Json?       // 自定义响应头
  
  route         RouteConfig @relation(fields: [routeId], references: [id], onDelete: Cascade)
  
  @@map("oss_configs")
}

enum ConfigType {
  REDIRECT // 重定向
  OSS      // OSS映射
}

enum RouteStatus {
  ACTIVE   // 活跃
  INACTIVE // 停用
  TESTING  // 测试中
}

enum HttpMethod {
  GET
  POST
  PUT
  DELETE
  PATCH
  HEAD
  OPTIONS
  ALL      // 匹配所有方法
}
```

### 3.3 权限管理表

```prisma
// 用户表（继承现有User表）
// model User {
//   id            String          @id @default(cuid())
//   name          String?
//   email         String          @unique
//   emailVerified DateTime?
//   image         String?
//   accounts      Account[]
//   sessions      Session[]
//   Authenticator Authenticator[]
//   
//   // 新增权限关联
//   appPermissions AppPermission[]
//   
//   createdAt DateTime @default(now())
//   updatedAt DateTime @updatedAt

// 应用权限表
model AppPermission {
  id            String            @id @default(cuid())
  userId        String            // 用户ID
  applicationId String            // 应用ID
  role          AppRole           // 角色
  permissions   PermissionType[]  // 权限列表
  
  // 关联关系
  user          User              @relation(fields: [userId], references: [id], onDelete: Cascade)
  application   Application       @relation(fields: [applicationId], references: [id], onDelete: Cascade)
  
  createdAt     DateTime          @default(now())
  updatedAt     DateTime          @updatedAt
  grantedBy     String?           // 授权者ID
  
  @@unique([userId, applicationId]) // 用户在每个应用下只能有一个权限记录
  @@map("app_permissions")
}

enum AppRole {
  OWNER    // 拥有者
  ADMIN    // 管理员
  EDITOR   // 编辑者
  VIEWER   // 查看者
}

enum PermissionType {
  READ     // 读取
  WRITE    // 写入
  DELETE   // 删除
  DEPLOY   // 部署
  ROLLBACK // 回滚
  MANAGE   // 管理权限
}
```

### 3.4 版本控制表

```prisma
// 配置版本表
model ConfigVersion {
  id            String        @id @default(cuid())
  applicationId String        // 应用ID
  routeId       String?       // 路由配置ID（为空表示应用级版本）
  version       String        // 版本号
  action        VersionAction // 操作类型
  description   String?       // 版本描述
  configData    Json          // 配置数据快照
  
  // 关联关系
  application   Application   @relation(fields: [applicationId], references: [id], onDelete: Cascade)
  route         RouteConfig?  @relation(fields: [routeId], references: [id], onDelete: Cascade)
  
  createdAt     DateTime      @default(now())
  createdBy     String        // 操作者ID
  
  @@map("config_versions")
}

// 操作日志表
model AuditLog {
  id            String        @id @default(cuid())
  userId        String        // 操作用户ID
  applicationId String        // 应用ID
  action        String        // 操作类型
  resource      String        // 操作资源
  resourceId    String        // 资源ID
  details       Json?         // 操作详情
  ipAddress     String?       // IP地址
  userAgent     String?       // 用户代理
  
  createdAt     DateTime      @default(now())
  
  @@map("audit_logs")
}

enum VersionAction {
  CREATE   // 创建
  UPDATE   // 更新
  DELETE   // 删除
  ROLLBACK // 回滚
  DEPLOY   // 部署
}
```

## 5. 数据关系图

```
Application (应用)
├── RouteConfig (路由配置)
│   ├── RedirectConfig (重定向配置)
│   └── OssConfig (OSS配置)
├── AppPermission (应用权限)
├── ConfigVersion (配置版本)
└── AuditLog (审计日志)

User (用户)
├── AppPermission (应用权限)
└── AuditLog (审计日志)
```

## 6. 索引优化建议

```sql
-- 应用查询优化
CREATE INDEX idx_applications_domain ON applications(domain);
CREATE INDEX idx_applications_status ON applications(status);

-- 路由配置查询优化
CREATE INDEX idx_route_configs_app_path ON route_configs(applicationId, path);
CREATE INDEX idx_route_configs_status ON route_configs(status);
CREATE INDEX idx_route_configs_priority ON route_configs(priority DESC);

-- 权限查询优化
CREATE INDEX idx_app_permissions_user ON app_permissions(userId);
CREATE INDEX idx_app_permissions_app ON app_permissions(applicationId);

-- 版本查询优化
CREATE INDEX idx_config_versions_app_time ON config_versions(applicationId, createdAt DESC);
CREATE INDEX idx_audit_logs_app_time ON audit_logs(applicationId, createdAt DESC);
```

## 7. 特性说明

### 7.1 权限控制
- 基于角色的权限控制（RBAC）
- 应用级权限隔离
- 细粒度权限控制（读、写、删除、部署、回滚等）

### 7.2 版本管理
- 配置变更自动版本化
- 支持任意版本回滚
- 完整的操作审计日志

### 7.3 配置类型
- 重定向配置：支持状态码、查询参数转发等
- OSS映射配置：支持多种OSS参数配置

### 7.4 扩展性
- 支持新的配置类型扩展
- 权限系统可扩展新的权限类型
- 版本系统支持应用级和路由级版本管理
